Question 1: Investigate if sex and island affect the body mass

## Data wrangling
df <- penguins |>
    drop_na() |>
    group_by(sex, island) |>
    summarize(mean_mass = mean(body_mass_g, na.rm = TRUE))

### Code revised from docs website
p2 <- ggplot(penguins, aes(x = bill_length_mm, y = flipper_length_mm, color = species)) +

    geom_smooth(method = "lm", se = FALSE) +
    scale_color_brewer(palette = "Set1") +
    labs(
        title = "Bill Length vs. Flipper Length",
